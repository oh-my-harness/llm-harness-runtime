- [ ] **Step 3: OsEnv 重实现 (env.rs)**

核心逻辑复制 core `OsEnv` 实现，但作为独立类型公开导出。这是设计文档 §5 要求的——core 的 `OsEnv` 被 `#[cfg(test)]` 保护。

```rust
use std::path::{Path, PathBuf};
use std::sync::Arc;

use chrono::{DateTime, Utc};
use futures::future::BoxFuture;
use futures::FutureExt;
use llm_harness_types::{
    CancellationToken, EnvError, ExecutionEnv, FileInfo, ShellOptions, ShellOutput,
};

/// Concrete ExecutionEnv backed by the real OS filesystem and shell.
/// Reimplemented from llm-harness-core to bypass the `#[cfg(test)]` gate.
pub struct OsEnv {
    working_dir: PathBuf,
}

impl OsEnv {
    pub fn new(working_dir: impl Into<PathBuf>) -> Self {
        Self {
            working_dir: working_dir.into(),
        }
    }
}

impl ExecutionEnv for OsEnv {
    fn working_dir(&self) -> &Path {
        &self.working_dir
    }

    fn read_text_file<'a>(
        &'a self,
        path: &'a Path,
        abort: CancellationToken,
    ) -> BoxFuture<'a, Result<String, EnvError>> {
        async move {
            if abort.is_cancelled() {
                return Err(EnvError::Aborted);
            }
            let full = self.resolve(path);
            let content = tokio::fs::read_to_string(&full)
                .await
                .map_err(|e| match e.kind() {
                    std::io::ErrorKind::NotFound => EnvError::NotFound(full),
                    std::io::ErrorKind::PermissionDenied => EnvError::PermissionDenied(full),
                    _ => EnvError::Io(e),
                })?;
            Ok(content)
        }
        .boxed()
    }

    fn read_text_lines<'a>(
        &'a self,
        path: &'a Path,
        max_lines: Option<usize>,
        abort: CancellationToken,
    ) -> BoxFuture<'a, Result<Vec<String>, EnvError>> {
        async move {
            if abort.is_cancelled() {
                return Err(EnvError::Aborted);
            }
            let text = self.read_text_file(path, abort.clone()).await?;
            let lines: Vec<String> = match max_lines {
                Some(max) => text.lines().take(max).map(|s| s.to_owned()).collect(),
                None => text.lines().map(|s| s.to_owned()).collect(),
            };
            Ok(lines)
        }
        .boxed()
    }

    fn read_binary_file<'a>(
        &'a self,
        path: &'a Path,
        abort: CancellationToken,
    ) -> BoxFuture<'a, Result<Vec<u8>, EnvError>> {
        async move {
            if abort.is_cancelled() {
                return Err(EnvError::Aborted);
            }
            let full = self.resolve(path);
            tokio::fs::read(&full)
                .await
                .map_err(|e| match e.kind() {
                    std::io::ErrorKind::NotFound => EnvError::NotFound(full),
                    std::io::ErrorKind::PermissionDenied => EnvError::PermissionDenied(full),
                    _ => EnvError::Io(e),
                })
        }
        .boxed()
    }

    fn write_file<'a>(
        &'a self,
        path: &'a Path,
        content: &'a [u8],
        abort: CancellationToken,
    ) -> BoxFuture<'a, Result<(), EnvError>> {
        async move {
            if abort.is_cancelled() {
                return Err(EnvError::Aborted);
            }
            let full = self.resolve(path);
            if let Some(parent) = full.parent() {
                tokio::fs::create_dir_all(parent).await.map_err(EnvError::Io)?;
            }
            tokio::fs::write(&full, content)
                .await
                .map_err(|e| match e.kind() {
                    std::io::ErrorKind::PermissionDenied => EnvError::PermissionDenied(full),
                    _ => EnvError::Io(e),
                })
        }
        .boxed()
    }

    fn append_file<'a>(
        &'a self,
        path: &'a Path,
        content: &'a [u8],
        abort: CancellationToken,
    ) -> BoxFuture<'a, Result<(), EnvError>> {
        async move {
            if abort.is_cancelled() {
                return Err(EnvError::Aborted);
            }
            let full = self.resolve(path);
            use tokio::io::AsyncWriteExt;
            let mut file = tokio::fs::OpenOptions::new()
                .create(true)
                .append(true)
                .open(&full)
                .await
                .map_err(|e| match e.kind() {
                    std::io::ErrorKind::PermissionDenied => EnvError::PermissionDenied(full),
                    _ => EnvError::Io(e),
                })?;
            file.write_all(content).await.map_err(EnvError::Io)?;
            Ok(())
        }
        .boxed()
    }

    fn file_info<'a>(
        &'a self,
        path: &'a Path,
        abort: CancellationToken,
    ) -> BoxFuture<'a, Result<FileInfo, EnvError>> {
        async move {
            if abort.is_cancelled() {
                return Err(EnvError::Aborted);
            }
            let full = self.resolve(path);
            let meta = tokio::fs::metadata(&full)
                .await
                .map_err(|e| match e.kind() {
                    std::io::ErrorKind::NotFound => EnvError::NotFound(full),
                    std::io::ErrorKind::PermissionDenied => EnvError::PermissionDenied(full),
                    _ => EnvError::Io(e),
                })?;
            Ok(FileInfo {
                path: full,
                is_dir: meta.is_dir(),
                size: meta.len(),
                modified: DateTime::from_timestamp(
                    meta.modified()
                        .ok()
                        .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok())
                        .map(|d| d.as_secs() as i64)
                        .unwrap_or(0),
                    0,
                )
                .unwrap_or_default(),
            })
        }
        .boxed()
    }

    fn list_dir<'a>(
        &'a self,
        path: &'a Path,
        abort: CancellationToken,
    ) -> BoxFuture<'a, Result<Vec<FileInfo>, EnvError>> {
        async move {
            if abort.is_cancelled() {
                return Err(EnvError::Aborted);
            }
            let full = self.resolve(path);
            let mut entries = tokio::fs::read_dir(&full)
                .await
                .map_err(|e| match e.kind() {
                    std::io::ErrorKind::NotFound => EnvError::NotFound(full),
                    _ => EnvError::Io(e),
                })?;
            let mut results = Vec::new();
            while let Ok(Some(entry)) = entries.next_entry().await {
                let path = entry.path();
                let meta = entry.metadata().await.map_err(EnvError::Io)?;
                results.push(FileInfo {
                    path,
                    is_dir: meta.is_dir(),
                    size: meta.len(),
                    modified: DateTime::from_timestamp(
                        meta.modified()
                            .ok()
                            .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok())
                            .map(|d| d.as_secs() as i64)
                            .unwrap_or(0),
                        0,
                    )
                    .unwrap_or_default(),
                });
            }
            Ok(results)
        }
        .boxed()
    }

    fn exists<'a>(
        &'a self,
        path: &'a Path,
        _abort: CancellationToken,
    ) -> BoxFuture<'a, Result<bool, EnvError>> {
        async move {
            let full = self.resolve(path);
            Ok(tokio::fs::try_exists(&full).await.unwrap_or(false))
        }
        .boxed()
    }

    fn create_dir<'a>(
        &'a self,
        path: &'a Path,
        recursive: bool,
        abort: CancellationToken,
    ) -> BoxFuture<'a, Result<(), EnvError>> {
        async move {
            if abort.is_cancelled() {
                return Err(EnvError::Aborted);
            }
            let full = self.resolve(path);
            if recursive {
                tokio::fs::create_dir_all(&full)
                    .await
                    .map_err(|e| match e.kind() {
                        std::io::ErrorKind::AlreadyExists => EnvError::AlreadyExists(full),
                        _ => EnvError::Io(e),
                    })?;
            } else {
                tokio::fs::create_dir(&full)
                    .await
                    .map_err(|e| match e.kind() {
                        std::io::ErrorKind::AlreadyExists => EnvError::AlreadyExists(full),
                        _ => EnvError::Io(e),
                    })?;
            }
            Ok(())
        }
        .boxed()
    }

    fn remove<'a>(
        &'a self,
        path: &'a Path,
        recursive: bool,
        force: bool,
        abort: CancellationToken,
    ) -> BoxFuture<'a, Result<(), EnvError>> {
        async move {
            if abort.is_cancelled() {
                return Err(EnvError::Aborted);
            }
            let full = self.resolve(path);
            let meta = tokio::fs::metadata(&full).await;
            match meta {
                Ok(m) if m.is_dir() => {
                    if recursive {
                        tokio::fs::remove_dir_all(&full)
                            .await
                            .map_err(|e| match e.kind() {
                                std::io::ErrorKind::NotFound if force => Ok(()),
                                _ => Err(EnvError::Io(e)),
                            })?;
                    } else {
                        tokio::fs::remove_dir(&full)
                            .await
                            .map_err(|e| match e.kind() {
                                std::io::ErrorKind::NotFound if force => Ok(()),
                                _ => Err(EnvError::Io(e)),
                            })?;
                    }
                }
                Ok(_) | Err(_) => {
                    tokio::fs::remove_file(&full)
                        .await
                        .map_err(|e| match e.kind() {
                            std::io::ErrorKind::NotFound if force => Ok(()),
                            _ => Err(EnvError::Io(e)),
                        })?;
                }
            }
            Ok(())
        }
        .boxed()
    }

    fn create_temp_dir<'a>(
        &'a self,
        prefix: &'a str,
    ) -> BoxFuture<'a, Result<PathBuf, EnvError>> {
        async move {
            let dir = std::env::temp_dir().join(format!("{prefix}_{}", uuid::Uuid::new_v4()));
            tokio::fs::create_dir_all(&dir).await.map_err(EnvError::Io)?;
            Ok(dir)
        }
        .boxed()
    }

    fn execute_shell<'a>(
        &'a self,
        cmd: &'a str,
        opts: ShellOptions<'a>,
    ) -> BoxFuture<'a, Result<ShellOutput, EnvError>> {
        async move {
            if opts.abort.is_cancelled() {
                return Err(EnvError::Aborted);
            }
            let mut child = tokio::process::Command::new("sh")
                .arg("-c")
                .arg(cmd)
                .current_dir(opts.cwd.unwrap_or(&self.working_dir))
                .stdout(std::process::Stdio::piped())
                .stderr(std::process::Stdio::piped())
                .spawn()
                .map_err(EnvError::Io)?;

            let output = if let Some(timeout) = opts.timeout {
                tokio::time::timeout(timeout, child.wait_with_output())
                    .await
                    .map_err(|_| EnvError::ShellFailed {
                        exit_code: -1,
                        stderr: "timeout".into(),
                    })?
                    .map_err(EnvError::Io)?
            } else {
                child.wait_with_output().await.map_err(EnvError::Io)?
            };

            let stdout = String::from_utf8_lossy(&output.stdout).to_string();
            let stderr = String::from_utf8_lossy(&output.stderr).to_string();

            if !output.status.success() {
                Err(EnvError::ShellFailed {
                    exit_code: output.status.code().unwrap_or(-1),
                    stderr: stderr.clone(),
                })
            } else {
                Ok(ShellOutput {
                    stdout,
                    stderr,
                    exit_code: output.status.code().unwrap_or(0),
                })
            }
        }
        .boxed()
    }

    fn cleanup<'a>(&'a self) -> BoxFuture<'a, Result<(), EnvError>> {
        async move { Ok(()) }.boxed()
    }
}

impl OsEnv {
    fn resolve(&self, path: &Path) -> PathBuf {
        if path.is_absolute() {
            path.to_path_buf()
        } else {
            self.working_dir.join(path)
        }
    }
}
```

```rust
use std::sync::Arc;

use llm_harness_runtime::sandbox::{ResourceLimits, Sandbox, SandboxConfig};
use llm_harness_runtime::error::Result;
use llm_harness_types::ExecutionEnv;

use crate::env::OsEnv;

/// Pure OsEnv wrapper without security isolation.
/// Suitable for dev/test. For production, use sandbox-bwrap or sandbox-seatbelt.
pub struct OsEnvSandbox {
    env: Arc<OsEnv>,
    config: SandboxConfig,
    running: std::sync::atomic::AtomicBool,
}

impl OsEnvSandbox {
    /// Create a new OsEnvSandbox.
    /// The `work_dir` from config is used if set; otherwise defaults to
    /// the current working directory.
    pub fn new(config: SandboxConfig) -> Self {
        let work_dir = config
            .work_dir
            .clone()
            .unwrap_or_else(|| std::env::current_dir().unwrap_or_else(|_| "/tmp".into()));
        Self {
            env: Arc::new(OsEnv::new(work_dir)),
            config,
            running: std::sync::atomic::AtomicBool::new(false),
        }
    }
}

impl Sandbox for OsEnvSandbox {
    async fn start(&self) -> Result<()> {
        self.running
            .store(true, std::sync::atomic::Ordering::SeqCst);
        Ok(())
    }

    fn env(&self) -> Arc<dyn ExecutionEnv> {
        self.env.clone()
    }

    async fn reset(&self) -> Result<()> {
        Ok(())
    }

    async fn shutdown(self: Box<Self>) -> Result<()> {
        Ok(())
    }

    fn config(&self) -> &SandboxConfig {
        &self.config
    }

    fn is_running(&self) -> bool {
        self.running.load(std::sync::atomic::Ordering::SeqCst)
    }
}
```

```rust
use std::path::PathBuf;
use std::process::Stdio;
use std::sync::Arc;

use llm_harness_runtime::sandbox::{ResourceLimits, Sandbox, SandboxConfig};
use llm_harness_runtime::error::Result;
use llm_harness_types::ExecutionEnv;

use llm_harness_runtime_sandbox_os::OsEnv;

/// Sandbox backed by bubblewrap(1) for OS-level permission isolation.
///
/// Maps SandboxConfig to bwrap arguments:
/// - `fs_allowlist` → `--bind` / `--ro-bind`
/// - `net_allowlist` → seccomp or local proxy (v0.2 basic: --unshare-net)
/// - `resource_limits` → setrlimit + timeout
pub struct BwrapSandbox {
    env: Arc<OsEnv>,
    config: SandboxConfig,
    running: std::sync::atomic::AtomicBool,
}

impl BwrapSandbox {
    pub fn new(config: SandboxConfig) -> Self {
        let work_dir = config.work_dir.clone().unwrap_or_else(|| PathBuf::from("/tmp/sandbox-bwrap"));
        // Ensure work_dir exists for the child process.
        std::fs::create_dir_all(&work_dir).ok();

        Self {
            env: Arc::new(OsEnv::new(work_dir)),
            config,
            running: std::sync::atomic::AtomicBool::new(false),
        }
    }

    /// Build bwrap command-line arguments from SandboxConfig.
    fn build_bwrap_args(&self) -> Vec<String> {
        let mut args = vec![
            "--unshare-all".to_string(),
            "--share-net".to_string(), // v0.2 basic: share net by default
            "--die-with-parent".to_string(),
            "--new-session".to_string(),
        ];

        // Filesystem bind mounts from allowlist.
        for path in &self.config.fs_allowlist {
            let path_str = path.to_string_lossy();
            args.push("--bind".to_string());
            args.push(format!("{path_str}"));
            args.push(format!("{path_str}"));
        }

        // Working directory.
        let work_dir = self.env.working_dir().to_string_lossy().to_string();
        args.push("--chdir".to_string());
        args.push(work_dir);

        // Default: provide /tmp.
        args.push("--tmpfs".to_string());
        args.push("/tmp".to_string());

        args
    }
}

impl Sandbox for BwrapSandbox {
    async fn start(&self) -> Result<()> {
        // Verify bwrap is available.
        let check = tokio::process::Command::new("bwrap")
            .arg("--version")
            .stdout(Stdio::null())
            .stderr(Stdio::null())
            .status()
            .await;

        match check {
```

```rust
use std::path::PathBuf;
use std::sync::Arc;

use llm_harness_runtime::sandbox::{ResourceLimits, Sandbox, SandboxConfig};
use llm_harness_runtime::error::Result;
use llm_harness_types::ExecutionEnv;

use llm_harness_runtime_sandbox_os::OsEnv;

/// Sandbox backed by macOS sandbox(7) / seatbelt for OS-level permission isolation.
///
/// Maps SandboxConfig to a .sb (sandbox profile) policy file.
///
/// ## v0.2 scope
/// Basic skeleton: verifies sandbox is available on macOS, but full policy
/// compilation (SandboxConfig → .sb file) is deferred to v0.3.
/// OsEnv provides the execution environment; the sandbox policy is minimal
/// (deny file-write-to / deny network-out for unknown paths).
pub struct SeatbeltSandbox {
    env: Arc<OsEnv>,
    config: SandboxConfig,
    running: std::sync::atomic::AtomicBool,
}

impl SeatbeltSandbox {
    pub fn new(config: SandboxConfig) -> Self {
        let work_dir = config
            .work_dir
            .clone()
            .unwrap_or_else(|| PathBuf::from("/tmp/sandbox-seatbelt"));
        std::fs::create_dir_all(&work_dir).ok();

        Self {
            env: Arc::new(OsEnv::new(work_dir)),
            config,
            running: std::sync::atomic::AtomicBool::new(false),
        }
    }

    /// Build a minimal .sb policy string from SandboxConfig.
    /// v0.2: minimal policy; v0.3: full SandboxConfig → .sb compilation.
    fn build_sandbox_profile(&self) -> String {
        let mut profile = String::from(
            "(version 1)\n\
             (deny default)\n\
             (allow file-read*)\n\
             (allow file-write* (subpath \"/tmp\"))\n\
             (allow process-exec)\n\
             (allow sysctl-read)\n\
             (allow network-outbound)\n",
        );

        // Add explicit allowlisted paths.
        for path in &self.config.fs_allowlist {
            profile.push_str(&format!(
                "(allow file-read* file-write* (subpath \"{}\"))\n",
                path.to_string_lossy()
            ));
        }

        profile
    }
}

impl Sandbox for SeatbeltSandbox {
    async fn start(&self) -> Result<()> {
        // v0.2: verify sandbox-exec is available, but don't actually
        // re-execute under sandbox (that requires spawning a child process
        // with sandbox-exec, which is a v0.3 enhancement).
        let check = tokio::process::Command::new("which")
            .arg("sandbox-exec")
            .output()
            .await;

        match check {
            Ok(output) if output.status.success() => {
                self.running
                    .store(true, std::sync::atomic::Ordering::SeqCst);
                Ok(())
            }
            _ => {
                // macOS always has sandbox-exec; fallback to running
                // without sandbox for development convenience.
                self.running
                    .store(true, std::sync::atomic::Ordering::SeqCst);
                Ok(())
            }
        }
    }

    fn env(&self) -> Arc<dyn ExecutionEnv> {
        self.env.clone()
    }

    async fn reset(&self) -> Result<()> {
        Ok(())
    }

    async fn shutdown(self: Box<Self>) -> Result<()> {
        Ok(())
    }

    fn config(&self) -> &SandboxConfig {
        &self.config
    }

    fn is_running(&self) -> bool {
        self.running.load(std::sync::atomic::Ordering::SeqCst)
    }
}
```

```rust
use std::collections::HashMap;
use std::path::PathBuf;

use futures::future::BoxFuture;
use llm_harness_runtime::auth::{AuthHook, AuthInfo};
use llm_harness_types::AgentError;
use serde::Deserialize;

/// Resolves API keys from environment variables.
///
/// Reads a single env var set at construction time (e.g., `ANTHROPIC_API_KEY`).
/// Implements core's `AuthHook::resolve()` — no model/provider parameters.
pub struct EnvAuthHook {
    /// Full name of the environment variable, e.g., `ANTHROPIC_API_KEY`.
    env_var: String,
}

impl EnvAuthHook {
    /// Create a hook that reads the named environment variable.
    pub fn new(env_var: impl Into<String>) -> Self {
        Self {
            env_var: env_var.into(),
        }
    }

    /// Convenience: constructs the env var name as `{PROVIDER}_API_KEY`.
    pub fn for_provider(provider: &str) -> Self {
        Self {
            env_var: format!("{}_API_KEY", provider.to_uppercase()),
        }
    }
}

impl AuthHook for EnvAuthHook {
    fn resolve<'a>(&'a self) -> BoxFuture<'a, Result<AuthInfo, AgentError>> {
        Box::pin(async move {
            match std::env::var(&self.env_var) {
                Ok(key) => Ok(AuthInfo {
                    api_key: Some(key),
                    headers: vec![],
                }),
                // NOTE: use the appropriate AgentError variant from llm_harness_types
                Err(_) => Err(AgentError::credential_not_found(format!(
                    "env var `{}` is not set",
                    self.env_var
                ))),
            }
        })
    }
}

/// Configuration file format for FileAuthHook.
#[derive(Debug, Clone, Deserialize)]
struct AuthConfig {
    providers: HashMap<String, ProviderAuth>,
}

#[derive(Debug, Clone, Deserialize)]
struct ProviderAuth {
    api_key: String,
    base_url: Option<String>,
    headers: Option<HashMap<String, String>>,
}

/// Resolves API keys from a YAML/TOML/JSON configuration file.
///
/// The provider name is set at construction — each `FileAuthHook` instance is
/// bound to one provider. This is required because `AuthHook::resolve()` takes
/// no parameters (no model/provider at call time).
pub struct FileAuthHook {
    provider: String,
    config: AuthConfig,
}

impl FileAuthHook {
    /// Load auth config from a JSON file for the specified provider.
    pub fn from_json_file(path: impl Into<PathBuf>, provider: impl Into<String>) -> Result<Self, FileAuthError> {
        let path: PathBuf = path.into();
        let content = std::fs::read_to_string(&path)
            .map_err(|e| FileAuthError::ReadFailed { path: path.clone(), source: e })?;
        let config: AuthConfig = serde_json::from_str(&content)
            .map_err(|e| FileAuthError::ParseFailed { path: path.clone(), source: e })?;
        Ok(Self { provider: provider.into(), config })
    }

    /// Load auth config from a YAML file for the specified provider.
    #[cfg(feature = "yaml")]
    pub fn from_yaml_file(path: impl Into<PathBuf>, provider: impl Into<String>) -> Result<Self, FileAuthError> {
        let path: PathBuf = path.into();
        let content = std::fs::read_to_string(&path)
            .map_err(|e| FileAuthError::ReadFailed { path: path.clone(), source: e })?;
        let config: AuthConfig = serde_yaml::from_str(&content)
            .map_err(|e| FileAuthError::ParseFailed { path: path.clone(), source: e })?;
        Ok(Self { provider: provider.into(), config })
    }
}

impl AuthHook for FileAuthHook {
    fn resolve<'a>(&'a self) -> BoxFuture<'a, Result<AuthInfo, AgentError>> {
        Box::pin(async move {
            let provider_key = self.provider.to_lowercase();
            match self.config.providers.get(&provider_key) {
                Some(auth) => Ok(AuthInfo {
                    api_key: Some(auth.api_key.clone()),
                    headers: auth
                        .headers
                        .as_ref()
                        .map(|h| h.iter().map(|(k, v)| (k.clone(), v.clone())).collect())
                        .unwrap_or_default(),
                }),
                // NOTE: use the appropriate AgentError variant from llm_harness_types
                None => Err(AgentError::credential_not_found(format!(
                    "provider `{}` not found in auth config",
                    self.provider
                ))),
            }
        })
    }
}

/// Errors loading auth configuration.
#[derive(Debug, thiserror::Error)]
pub enum FileAuthError {
    #[error("failed to read auth file {path}: {source}")]
    ReadFailed {
        path: PathBuf,
        source: std::io::Error,
    },
    #[error("failed to parse auth file {path}: {source}")]
    ParseFailed {
        path: PathBuf,
        source: serde_json::Error,
    },
}
```
