```rust
use std::path::PathBuf;
use tokio::sync::Mutex;    // tokio::sync::Mutex — 避免在持锁期间 .await 触发 UB

use llm_harness_runtime::audit::{AuditEntry, AuditSink};
use llm_harness_runtime::error::Result;
use sha2::{Digest, Sha256};
use tokio::io::AsyncWriteExt;
use tokio::fs::OpenOptions;

/// JSONL file-backed audit sink with hash-chain integrity protection.
///
/// Each entry is written as one JSON line. The hash chain is computed as:
///   hash_n = SHA256(serialize(content_n) + ":" + hash_{n-1})
///
/// Tampering is detectable when the verifier holds an independent copy
/// of the log (v0.2 limitation: no HMAC signing; coming in v0.3+).
pub struct JsonlAuditSink {
    path: PathBuf,
    prev_hash: Mutex<Option<String>>,
    /// tokio::sync::Mutex 允许在持锁期间 .await（std::sync::Mutex 不允许）。
    /// file 懒加载：第一次 record() 时打开。
    file: Mutex<Option<tokio::fs::File>>,
}

impl JsonlAuditSink {
    /// Create a new JSONL audit sink that writes to the given path.
    /// The file is lazily opened on first write.
    pub fn new(path: impl Into<PathBuf>) -> Self {
        Self {
            path: path.into(),
            prev_hash: Mutex::new(None),
            file: Mutex::new(None),
        }
    }

    /// Compute the hash of a serialized audit entry.
    fn hash_entry(content: &str, prev_hash: Option<&str>) -> String {
        let input = match prev_hash {
            Some(prev) => format!("{content}:{prev}"),
            None => content.to_string(),
        };
        let hash = Sha256::digest(input.as_bytes());
        hex::encode(hash)
    }

    /// Validate the hash chain of a JSONL audit log.
    /// Returns the number of valid entries, or an error on the first broken link.
    pub async fn validate(path: &std::path::Path) -> Result<usize> {
        let content = tokio::fs::read_to_string(path).await?;
        let mut count = 0;
        let mut prev_hash: Option<String> = None;

        for line in content.lines() {
            if line.trim().is_empty() {
                continue;
            }
            let entry: serde_json::Value =
                serde_json::from_str(line).map_err(|e| anyhow::anyhow!("parse error at line {count}: {e}"))?;

            let stored_hash = entry["hash"]
                .as_str()
                .ok_or_else(|| anyhow::anyhow!("missing hash at line {count}"))?;

            // Build content for re-hashing: everything except the "hash" field.
            let content_for_hash = {
                let mut stripped = entry.clone();
                stripped.as_object_mut().unwrap().remove("hash");
                serde_json::to_string(&stripped)?
            };

            let computed = Self::hash_entry(&content_for_hash, prev_hash.as_deref());

            if computed != stored_hash {
                return Err(anyhow::anyhow!(
                    "hash chain broken at line {count}: expected {stored_hash}, computed {computed}"
                ));
            }

            prev_hash = Some(computed);
            count += 1;
        }

        Ok(count)
    }
}

impl AuditSink for JsonlAuditSink {
    async fn record(&self, mut entry: AuditEntry) -> Result<()> {
        // Compute hash chain.
        let prev = self.prev_hash.lock().await.clone();
        entry.prev_hash = prev.clone();

        // Serialize content without hash field first.
        let content_for_hash = serde_json::to_value(&entry)?;
        let hash = Self::hash_entry(
            &serde_json::to_string(&content_for_hash)?,
            prev.as_deref(),
        );
        entry.hash = hash.clone();

        // Serialize full entry with hash.
        let line = serde_json::to_string(&entry)?;

        // Lazily open file — use tokio::sync::Mutex so .await is safe while holding guard.
        let mut file_guard = self.file.lock().await;
        if file_guard.is_none() {
            let f = OpenOptions::new()
                .create(true)
                .append(true)
                .open(&self.path)
                .await?;
            *file_guard = Some(f);
        }
        let file = file_guard.as_mut().unwrap();

        file.write_all(line.as_bytes()).await?;
        file.write_all(b"\n").await?;
        file.flush().await?;

        // Update hash chain.
        *self.prev_hash.lock().await = Some(hash);

        Ok(())
    }

    async fn flush(&self) -> Result<()> {
        if let Some(ref mut file) = *self.file.lock().await {
            file.flush().await?;
        }
        Ok(())
    }
}

// hex encoding helper (avoid adding a hex crate for this simple use case).
mod hex {
    pub fn encode(bytes: impl AsRef<[u8]>) -> String {
        bytes.as_ref().iter().map(|b| format!("{b:02x}")).collect()
    }
}
```

```rust
use std::sync::{Arc, Mutex, RwLock};

use chrono::Utc;
use llm_harness::{AgentHarness, AgentHarnessOptions};
use llm_harness_types::EntryId;
use serde_json::Value;

use crate::budget::BudgetControlAdapter;
use crate::cost::CostAggregate;
use crate::human_approval::HumanApprovalWrapper;
use crate::sandbox::Sandbox;

/// In-memory retry policy for TaskRunner.
#[derive(Debug, Clone)]
pub struct RetryPolicy {
    pub max_retries: usize,
    pub backoff_ms: u64,
}

impl Default for RetryPolicy {
    fn default() -> Self {
        Self {
            max_retries: 3,
            backoff_ms: 1000,
        }
    }
}

/// Concrete TaskRunner implementation managing task lifecycle.
///
/// One task may span multiple AgentHarness sessions (crash recovery,
/// checkpoint resume). The runner maintains task state, retries, and
/// checkpoints.
pub struct TaskRunnerImpl {
    task_id: TaskId,
    sandbox: Arc<dyn Sandbox>,
    verifier: Option<Arc<dyn TaskVerifier>>,
    retry_policy: RetryPolicy,
    budget_adapter: Option<Arc<BudgetControlAdapter>>,
    human_approval: Option<Arc<HumanApprovalWrapper>>,

    state: RwLock<TaskState>,
    cost: RwLock<CostAggregate>,
    checkpoints: Mutex<Vec<Checkpoint>>,
}

impl TaskRunnerImpl {
    /// Create a new TaskRunner.
    pub fn new(
        task_id: TaskId,
        sandbox: Arc<dyn Sandbox>,
        verifier: Option<Arc<dyn TaskVerifier>>,
        retry_policy: RetryPolicy,
        budget_adapter: Option<Arc<BudgetControlAdapter>>,
        human_approval: Option<Arc<HumanApprovalWrapper>>,
    ) -> Self {
        Self {
            task_id,
            sandbox,
            verifier,
            retry_policy,
            budget_adapter,
            human_approval,
            state: RwLock::new(TaskState::Pending),
            cost: RwLock::new(CostAggregate::default()),
            checkpoints: Mutex::new(Vec::new()),
        }
    }

    /// Transition state, returning the new state.
    fn transition(&self, new_state: TaskState) -> TaskState {
        let mut state = self.state.write().unwrap();
        *state = new_state.clone();
        new_state
    }

    /// Get the current budget adapter's cost estimate.
    pub fn session_cost(&self) -> Option<CostAggregate> {
        self.budget_adapter.as_ref().map(|b| b.current_cost())
    }
}

impl TaskRunner for TaskRunnerImpl {
    async fn start(&self, prompt: &str) -> Result<()> {
        // Transition to Running.
        let session_id = EntryId(uuid::Uuid::now_v7());
        let now = Utc::now();
        let state = self.transition(TaskState::Running {
            session_id,
            started_at: now,
        });

        // Build AgentHarness options using hooks.
        let mut harness_builder = AgentHarnessOptions::builder()
            .system_prompt("You are a helpful coding agent.")
            .build();

        // Inject hooks if configured.
        // In Phase 5, we'll actually wire hooks via HarnessHooks.
        // For now, we instantiate the harness with whatever hooks are available.

        // Create AgentHarness.
        let _harness = AgentHarness::new(harness_builder);

        // ⚠️ v0.2 限制：AgentHarness 的实际执行循环（prompt() 调用）在此处不运行。
        // TaskRunnerImpl 在 v0.2 中只提供状态机骨架（Pending→Running→Succeeded/Failed 等），
        // 不执行真实的 LLM 对话循环。
        //
        // 完整的 run loop 集成和端到端 E2E 测试推迟至 v0.2.1。
        // 3 个需要 mock LlmClient 的集成场景也在 v0.2.1 实现。
        // （参见设计文档 §7 Phase 5 的 E2E 覆盖范围说明）

        Ok(())
    }

    fn state(&self) -> TaskState {
        self.state.read().unwrap().clone()
    }

    async fn pause(&self, reason: &str) -> Result<()> {
        let current = self.state();
        match current {
            TaskState::Running { .. } => {
                self.transition(TaskState::Paused {
                    reason: reason.to_string(),
                });
                Ok(())
            }
            _ => Err(anyhow::anyhow!("cannot pause: task is not running")),
        }
    }

    async fn resume(&self) -> Result<()> {
        let current = self.state();
        match current {
            TaskState::Paused { .. } => {
                let session_id = EntryId(uuid::Uuid::now_v7());
                self.transition(TaskState::Running {
                    session_id,
                    started_at: Utc::now(),
                });
                Ok(())
            }
            _ => Err(anyhow::anyhow!("cannot resume: task is not paused")),
        }
    }

    async fn cancel(&self, reason: &str) -> Result<()> {
        self.transition(TaskState::Cancelled {
            reason: reason.to_string(),
        });
        Ok(())
    }

    fn total_cost(&self) -> CostAggregate {
        self.cost.read().unwrap().clone()
    }

    async fn checkpoint(&self, description: &str, payload: Value) -> Result<()> {
        let mut checkpoints = self.checkpoints.lock().unwrap();
        let seq = checkpoints.len();
        checkpoints.push(Checkpoint {
            seq,
            saved_at: Utc::now(),
            description: description.to_string(),
            payload,
        });
        Ok(())
    }
}
```

```rust
// Compute cost from token usage.
if let (Some(ref usage), Some(ref info_model), Some(ref info_provider)) = (
    &info.usage,
    info.model.as_ref(),
    info.provider.as_ref(),
) {
    let mut cost = self.session_cost.lock().unwrap();
    cost.total_input_tokens += usage.input_tokens as u64;
    cost.total_output_tokens += usage.output_tokens as u64;
    cost.total_cache_read_tokens += usage.cache_read_tokens as u64;
    cost.total_cache_write_tokens += usage.cache_creation_tokens as u64;

    // Look up pricing.
    if let Some(price) = self.pricing.price_for(info_model, info_provider) {
        let call_cost = (usage.input_tokens as f64 / 1_000_000.0) * price.input_per_mtok
            + (usage.output_tokens as f64 / 1_000_000.0) * price.output_per_mtok
            + (usage.cache_read_tokens as f64 / 1_000_000.0) * price.cache_read_per_mtok
            + (usage.cache_creation_tokens as f64 / 1_000_000.0) * price.cache_write_per_mtok;
        cost.total_cost += call_cost;

        // Update per-model cost.
        let model_key = format!("{info_provider}/{info_model}");
        let model_cost = cost.by_model.entry(model_key.clone()).or_default();
        model_cost.input_tokens += usage.input_tokens as u64;
        model_cost.output_tokens += usage.output_tokens as u64;
        model_cost.cost += call_cost;
        model_cost.call_count += 1;
    }

    // Check budget.
    if cost.total_cost > self.budget_limit {
        *self.budget_exceeded.lock().unwrap() = true;
        if let Some(ref hook) = self.exceeded_hook {
            let task_cost = CostAggregate::default(); // populated by TaskRunner
            let should_continue = hook
                .on_budget_exceeded(&cost, &task_cost, self.budget_limit)
                .await;
            *self.continue_decision.lock().unwrap() = should_continue;
        } else {
            // No hook = open loop; continue by default (cost surveillance only).
            *self.continue_decision.lock().unwrap() = true;
        }
    }
}
```
