use std::path::PathBuf;
use std::sync::Arc;
use std::time::Duration;

use llm_harness_types::ExecutionEnv;

use crate::error::Result;

/// Network access rule for sandbox configuration.
#[derive(Debug, Clone)]
pub struct NetRule {
    pub host: String,
    pub port_range: Option<(u16, u16)>,
}

/// Resource limits for sandbox execution.
#[derive(Debug, Clone)]
pub struct ResourceLimits {
    pub max_cpus: Option<usize>,
    pub max_memory_mb: Option<usize>,
    pub max_disk_mb: Option<usize>,
    pub timeout: Option<Duration>,
}

/// Sandbox permission boundary declaration (v0.2 minimal set).
#[derive(Debug, Clone)]
pub struct SandboxConfig {
    /// File system path allowlist (read/write).
    pub fs_allowlist: Vec<PathBuf>,
    /// File system path denylist (exceptions to the above allowlist).
    pub fs_denylist: Vec<PathBuf>,
    /// Network access rules.
    pub net_allowlist: Vec<NetRule>,
    /// Resource limits.
    pub resource_limits: ResourceLimits,
    /// Sandbox working directory.
    ///
    /// After `start()` succeeds, `env().working_dir()` MUST return this path
    /// (if set), otherwise the sandbox backend's default working directory.
    pub work_dir: Option<PathBuf>,
}

/// Sandbox abstraction — wraps ExecutionEnv, adding lifecycle management.
///
/// ## `work_dir` semantic contract
/// After `start()` succeeds, `env().working_dir()` returns:
/// - If `SandboxConfig.work_dir` is set → MUST return that path (backend should
///   set sandbox cwd to this path).
/// - If not set → returns the sandbox backend's default working directory
///   (e.g., `/tmp/sandbox-<id>` or `/home/agent`).
/// Backends MUST NOT silently modify `working_dir()` after `start()`.
pub trait Sandbox: Send + Sync {
    /// Start or verify the sandbox is ready.
    fn start(&self) -> impl std::future::Future<Output = Result<()>> + Send;

    /// Return the execution environment inside the sandbox.
    fn env(&self) -> Arc<dyn ExecutionEnv>;

    /// Reset to initial state.
    fn reset(&self) -> impl std::future::Future<Output = Result<()>> + Send;

    /// Shut down the sandbox, releasing resources.
    fn shutdown(self: Box<Self>) -> impl std::future::Future<Output = Result<()>> + Send;

    /// Return the current sandbox configuration.
    fn config(&self) -> &SandboxConfig;

    /// Whether the sandbox is currently running.
    fn is_running(&self) -> bool;
}
```

#[derive(Debug, Clone)]
pub struct ToolMetadata {

    pub label: String,
    pub description: String,

/// Tool registry — manages the full set of known tools.
pub trait ToolRegistry: Send + Sync {
    /// Register a tool.
    fn register(&self, tool: Arc<dyn Tool>);

    /// Unregister a tool by name.
    fn unregister(&self, name: &str);

    /// Look up a tool by name.
    fn get(&self, name: &str) -> Option<Arc<dyn Tool>>;

    fn list(&self) -> Vec<ToolMetadata>;

    /// Return a subset of tools by name list (coarse-grained, management-plane filter).
    fn subset(&self, names: &[&str]) -> Vec<Arc<dyn Tool>>;
}

/// Source of tool definitions (MCP server, local file, HTTP API, etc.).
pub trait ToolSource: Send + Sync {
    /// Discover and return available tool definitions.
    fn discover(&self) -> impl std::future::Future<Output = Result<Vec<ToolDefinition>>> + Send;

use serde::{Deserialize, Serialize};
use serde_json::Value;
use std::error::Error;

use crate::error::Result;

/// MCP client handle — traits that business layer implements.
/// Runtime does NOT bind to any specific MCP SDK.
pub trait McpClientHandle: Send + Sync {
    /// List tools from the MCP server (called once at startup).
    fn list_tools(&self) -> impl std::future::Future<Output = Result<Vec<McpToolDef>>> + Send;

    /// Call a tool on the MCP server.
    fn call_tool(
        &self,
        name: &str,
        arguments: Value,
    ) -> impl std::future::Future<Output = std::result::Result<McpResult, McpError>> + Send;
}

/// MCP tool definition as returned by tools/list.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct McpToolDef {
    pub name: String,
    pub description: String,
    pub input_schema: Value,
}

/// Result of an MCP tool call.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct McpResult {
    pub content: Vec<McpContent>,
    pub is_error: bool,
}

/// Content block within an MCP result.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct McpContent {
    #[serde(rename = "type")]
    pub content_type: String,
    pub text: Option<String>,
    pub data: Option<String>,
    pub mime_type: Option<String>,
}

/// MCP error — distinguishes protocol errors (infrastructure) from tool
/// execution errors (semantic failure the agent should retry).
#[derive(Debug)]
pub enum McpError {
    /// Protocol-level error: connection lost, timeout, etc. → escalate to runtime.
    Protocol {
        source: Box<dyn Error + Send>,
    },
    /// Tool execution failure → return to agent for retry.
    ToolExecution {
        content: Vec<McpContent>,
    },
}

impl std::fmt::Display for McpError {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        match self {
            McpError::Protocol { source } => write!(f, "MCP protocol error: {source}"),
            McpError::ToolExecution { .. } => write!(f, "MCP tool execution error"),
        }
    }
}

impl Error for McpError {
    fn source(&self) -> Option<&(dyn Error + 'static)> {
        match self {
            McpError::Protocol { source } => Some(source.as_ref()),
            McpError::ToolExecution { .. } => None,
        }
    }
}
```

```rust
use std::sync::Arc;

use llm_harness_types::{AgentContext, ContentBlock};

use crate::error::Result;

/// Description of a context resource.
#[derive(Debug, Clone)]
pub struct ResourceDescriptor {
    pub uri: String,
    pub name: String,
    pub description: String,
    pub mime_type: String,
}

/// Content of a context resource.
#[derive(Debug, Clone)]
pub enum ResourceContent {
    Text(String),
    Blob(Vec<u8>),
}

/// Context resource provider (MCP resources, local files, etc.).
pub trait ResourceProvider: Send + Sync {
    /// List available resources.
    fn list(&self) -> impl std::future::Future<Output = Result<Vec<ResourceDescriptor>>> + Send;

    /// Read a specific resource by URI.
    fn read(&self, uri: &str) -> impl std::future::Future<Output = Result<ResourceContent>> + Send;
}

/// Resource injector — appends resource content to system prompt tail
/// with cache_control marker for KV-cache friendliness.
pub struct ResourceInjector {
    // Injector uses core's AgentContext model to append resources.
    // The actual injection happens through TransformContextHook.
    _private: (),
}

impl ResourceInjector {
    /// Create a new injector.
    pub fn new() -> Self {
        Self { _private: () }
    }

    /// Inject all resources from all providers into the system prompt tail.
    /// Resources are appended with cache_control-compatible markers.
    pub async fn inject_all(
        &self,
        providers: &[Arc<dyn ResourceProvider>],
        context: &mut AgentContext,
    ) -> Result<()> {
        for provider in providers {
            let descriptors = provider.list().await?;
            for desc in descriptors {
                self.inject_one(provider.as_ref(), &desc.uri, context).await?;
            }
        }
        Ok(())
    }

    /// Inject a single resource by URI.
    pub async fn inject(
        &self,
        provider: &dyn ResourceProvider,
        uri: &str,
        context: &mut AgentContext,
    ) -> Result<()> {
        self.inject_one(provider, uri, context).await
    }

    async fn inject_one(
        &self,
        provider: &dyn ResourceProvider,
        uri: &str,
        context: &mut AgentContext,
    ) -> Result<()> {
        let content = provider.read(uri).await?;
        let text = match content {
            ResourceContent::Text(t) => t,
            ResourceContent::Blob(b) => format!("[binary resource, {} bytes]", b.len()),
        };

        // Append as system prompt tail with resource marker.
        let injected = format!(
            "<resource uri=\"{uri}\">\n{text}\n</resource>"
        );
        match &mut context.system_prompt {
            Some(prompt) => {
                prompt.push_str("\n\n");
                prompt.push_str(&injected);
            }
            None => {
                context.system_prompt = Some(injected);
            }
        }

        Ok(())
    }
}

impl Default for ResourceInjector {
    fn default() -> Self {
        Self::new()
    }
}
```

/// Argument definition for a prompt template parameter.
#[derive(Debug, Clone)]
pub struct ArgumentDef {
    pub name: String,
    pub description: String,
    pub required: bool,
}

/// Raw template produced by a PromptSource.
/// Uses `{{param_name}}` named placeholder syntax.
#[derive(Debug, Clone)]
pub struct PromptSourceTemplate {
    pub name: String,
    pub description: String,
    /// Template body with `{{param_name}}` named placeholders.
    pub template: String,
    /// Parameter definitions — order determines positional mapping
    /// when compiled to core PromptTemplate ($1, $2, ...).
    pub arguments: Vec<ArgumentDef>,
}

/// Source of prompt templates (MCP prompts, filesystem, etc.).
pub trait PromptSource: Send + Sync {
    /// Discover available templates.
    fn discover(&self) -> impl std::future::Future<Output = Result<Vec<PromptSourceTemplate>>> + Send;
}

/// Compiles PromptSourceTemplate into core PromptTemplate by mapping
/// `{{param_name}}` named placeholders to positional `$1`, `$2`, ...
/// based on `arguments` declaration order.
pub struct PromptTemplateCompiler;

impl PromptTemplateCompiler {
    /// Compile a PromptSourceTemplate into a core PromptTemplate.
    ///
    /// Example:
    ///   arguments = [ArgumentDef { name: "lang" }, ArgumentDef { name: "code" }]
    ///   "Review {{code}} in {{lang}}" → "Review $2 in $1"
    pub fn compile(source: &PromptSourceTemplate) -> std::result::Result<PromptTemplate, CompileError> {
        let mut template_str = source.template.clone();
        for (idx, arg) in source.arguments.iter().enumerate() {
            let placeholder = format!("{{{{{}}}}}", arg.name);
            let position = format!("${}", idx + 1);
            if !template_str.contains(&placeholder) {
                return Err(CompileError::MissingPlaceholder {
                    template: source.name.clone(),
                    param: arg.name.clone(),
                });
            }
            template_str = template_str.replace(&placeholder, &position);
        }
        // Check for any remaining unmatched {{...}} placeholders.
        if template_str.contains("{{") && template_str.contains("}}") {
            return Err(CompileError::UnmatchedPlaceholders {
                template: source.name.clone(),
                remaining: template_str.clone(),
            });
        }
        Ok(PromptTemplate::new(template_str))
    }
}

/// Errors during prompt template compilation.
#[derive(Debug, thiserror::Error)]
pub enum CompileError {
    #[error("template '{template}': parameter '{param}' declared but not found in template body")]
    MissingPlaceholder { template: String, param: String },
    #[error("template '{template}': unmatched {{...}} placeholders remain after compilation: {remaining}")]
    UnmatchedPlaceholders { template: String, remaining: String },
}

use uuid::Uuid;

/// Lightweight human approval interface (business layer implements).
/// Only concerns "asking the human" — does not involve core hook protocol.
pub trait HumanApprover: Send + Sync {
    /// Ask a human whether to authorize. Returns None on timeout/no response.
    fn ask(
        &self,
        request: &ApprovalRequest,
    ) -> impl std::future::Future<Output = Option<ApprovalDecision>> + Send;
}

/// Request for human approval — provides enough context to decide.
#[derive(Debug, Clone)]
pub struct ApprovalRequest {
    pub request_id: String,
    pub tool_name: String,
    pub arguments: Value,
    /// Sufficient context for the human to make a judgment
    /// (current task description, last step result, etc.).
    pub context: String,
}

/// Human's decision (minimal; allow-once/allow-always UX is maintained
/// by the business layer inside HumanApprover).
#[derive(Debug, Clone)]
pub enum ApprovalDecision {
    Approve,
    Deny { reason: String },
}

/// Timeout policy when HumanApprover::ask() returns None.
#[derive(Debug, Clone)]
pub enum TimeoutPolicy {
    Approve,
    Deny { reason: String },
}

/// Adapts HumanApprover into a core BeforeToolCallHook.
///
/// Default timeout policy is Deny (fail-secure): registering a HumanApprover
/// signals explicit need for human oversight; timeout = supervisor unavailable,
/// should block rather than pass through.
pub struct HumanApprovalWrapper {
    approver: Arc<dyn HumanApprover>,
    on_timeout: TimeoutPolicy,
}

impl HumanApprovalWrapper {
    /// Create a new wrapper with the given approver and timeout policy.
    /// Default `on_timeout` should be `TimeoutPolicy::Deny` for fail-secure.
    pub fn new(approver: Arc<dyn HumanApprover>, on_timeout: TimeoutPolicy) -> Self {
        Self {
            approver,
            on_timeout,
        }
    }
}

impl BeforeToolCallHook for HumanApprovalWrapper {
    fn on_call<'a>(
        &'a self,
        ctx: BeforeToolCallCtx<'a>,
    ) -> std::pin::Pin<Box<dyn std::future::Future<Output = BeforeToolCallDecision> + Send + 'a>> {
        Box::pin(async move {
            let request = ApprovalRequest {
                request_id: Uuid::new_v4().to_string(),
                tool_name: ctx.tool_name.to_owned(),
                arguments: ctx.args.clone(),
                context: String::new(), // populated by business-layer HumanApprover
            };
            match self.approver.ask(&request).await {
                Some(ApprovalDecision::Approve) => BeforeToolCallDecision::Allow,
                Some(ApprovalDecision::Deny { reason }) => {
                    BeforeToolCallDecision::Deny(ToolResult {
                        content: vec![],
                        details: serde_json::json!({"reason": reason}),
                        terminate: false,
                    })
                }
                None => match &self.on_timeout {
                    TimeoutPolicy::Approve => BeforeToolCallDecision::Allow,
                    TimeoutPolicy::Deny { reason } => {
                        BeforeToolCallDecision::Deny(ToolResult {
                            content: vec![],
                            details: serde_json::json!({"reason": reason}),
                            terminate: false,
                        })
                    }
                },
            }
        })
    }
}

use std::sync::{Arc, Mutex};

use llm_harness::{AfterProviderResponseHook, ShouldStopHook};
use llm_harness_types::{AgentError, ProviderResponseInfo};
use futures::future::BoxFuture;

use crate::cost::{CostAggregate, PricingProvider};

/// Business-layer hook for budget override decisions.
/// Called when the per-task budget limit is exceeded.
pub trait BudgetExceededHook: Send + Sync {
    /// Called when budget is exceeded; return true to continue, false to
    /// transition task to Failed.
    fn on_budget_exceeded(
        &self,
        session_cost: &CostAggregate,
        task_cost: &CostAggregate,
        budget_limit: f64,
    ) -> impl std::future::Future<Output = bool> + Send;
}

/// Adapts BudgetExceededHook into core hooks:
///   - AfterProviderResponseHook: accumulates cost, detects budget exceeded
///   - ShouldStopHook: returns true when budget_exceeded flag is set
///
/// NOTE: PrepareNextTurnHook cannot halt the loop — NextTurnDirective with
/// all-None fields means "continue normally". ShouldStopHook is the correct
/// core extension point for stopping the agent loop.
pub struct BudgetControlAdapter {
    pricing: Arc<dyn PricingProvider>,
    budget_limit: f64,
    exceeded_hook: Option<Arc<dyn BudgetExceededHook>>,
    session_cost: Mutex<CostAggregate>,
    /// True once the budget threshold has been crossed.
    /// Set by AfterProviderResponseHook; read by ShouldStopHook.
    budget_exceeded: Mutex<bool>,
    /// Stores the exceeded hook's decision: true = "continue despite over-budget",
    /// false = "stop now". Defaults to true (open loop until hook says stop).
    /// Phase 3 updates this field with the hook's return value.
    continue_decision: Mutex<bool>,
}

impl BudgetControlAdapter {
    /// Create a new adapter.
    /// `budget_limit` in dollars (e.g., 5.0 for $5).
    /// `exceeded_hook` of None means no budget control (open loop).
    pub fn new(
        pricing: Arc<dyn PricingProvider>,
        budget_limit: f64,
        exceeded_hook: Option<Arc<dyn BudgetExceededHook>>,
    ) -> Self {
        Self {
            pricing,
            budget_limit,
            exceeded_hook,
            session_cost: Mutex::new(CostAggregate::default()),
            budget_exceeded: Mutex::new(false),
            continue_decision: Mutex::new(true),
        }
    }

    /// Current session cost estimate.
    pub fn current_cost(&self) -> CostAggregate {
        self.session_cost.lock().unwrap().clone()
    }
}

impl AfterProviderResponseHook for BudgetControlAdapter {
    fn after_response<'a>(
        &'a self,
        info: &'a ProviderResponseInfo,
    ) -> BoxFuture<'a, ()> {
        Box::pin(async move {
            if let Some(ref usage) = info.usage {
                let mut cost = self.session_cost.lock().unwrap();
                cost.total_input_tokens  += usage.input_tokens as u64;
                cost.total_output_tokens += usage.output_tokens as u64;
                cost.total_cache_read_tokens  += usage.cache_read_tokens as u64;
                cost.total_cache_write_tokens += usage.cache_creation_tokens as u64;
            }

            // Check if budget exceeded (after updating cost).
            if let Some(ref hook) = self.exceeded_hook {
                let cost_snapshot = self.session_cost.lock().unwrap().clone();
                // Simple token-based estimate (real pricing via PricingProvider
                // is computed by TaskRunner which has model/provider context).
                if cost_snapshot.total_cost >= self.budget_limit {
                    let should_continue = hook
                        .on_budget_exceeded(
                            &cost_snapshot,
                            &cost_snapshot, // task_cost: same as session in v0.2
                            self.budget_limit,
                        )
                        .await;
                    if !should_continue {
                        *self.budget_exceeded.lock().unwrap() = true;
                    }
                }
            }
        })
    }
}

impl ShouldStopHook for BudgetControlAdapter {
    fn should_stop<'a>(&'a self) -> BoxFuture<'a, bool> {
        Box::pin(async move {
            // Stop only when budget is exceeded AND the hook decided to stop
            // (continue_decision = false). Without an exceeded_hook the adapter
            // is in cost-surveillance-only mode and never stops the loop.
            *self.budget_exceeded.lock().unwrap() && !*self.continue_decision.lock().unwrap()
        })
    }
}
```

    fn after_response<'a>(
        &'a self,
        info: &'a ProviderResponseInfo,
    ) -> BoxFuture<'a, ()> {
        Box::pin(async move {
            for hook in &self.hooks {
                hook.after_response(info).await;
            }
        })
    }
}

/// Chain multiple AfterToolCallHook: all execute in order.
pub struct CompositeAfterToolCallHook {
    hooks: Vec<Arc<dyn AfterToolCallHook>>,
}

impl CompositeAfterToolCallHook {
    pub fn new(hooks: Vec<Arc<dyn AfterToolCallHook>>) -> Self {
        Self { hooks }
    }
}

impl AfterToolCallHook for CompositeAfterToolCallHook {
    fn on_complete<'a>(
        &'a self,
        ctx: AfterToolCallCtx<'a>,
    ) -> BoxFuture<'a, AfterToolCallDecision> {
        Box::pin(async move {
            let mut decision = AfterToolCallDecision::Passthrough;
            for hook in &self.hooks {
                // Pass the same ctx; AfterToolCallDecision::Passthrough means no override.
                let d = hook.on_complete(ctx.clone()).await;
                if d != AfterToolCallDecision::Passthrough {
                    decision = d;
                }
            }
            decision
        })
    }
}

/// Chain multiple BeforeToolCallHook: short-circuits on first non-Allow decision.
pub struct CompositeBeforeToolCallHook {
    hooks: Vec<Arc<dyn BeforeToolCallHook>>,
}

impl CompositeBeforeToolCallHook {
    pub fn new(hooks: Vec<Arc<dyn BeforeToolCallHook>>) -> Self {
        Self { hooks }
    }
}

impl BeforeToolCallHook for CompositeBeforeToolCallHook {
    fn on_call<'a>(
        &'a self,
        ctx: BeforeToolCallCtx<'a>,
    ) -> BoxFuture<'a, BeforeToolCallDecision> {
        Box::pin(async move {
            for hook in &self.hooks {
                let decision = hook.on_call(ctx.clone()).await;
                if decision != BeforeToolCallDecision::Allow {
                    return decision; // short-circuit
                }
            }
            BeforeToolCallDecision::Allow
        })
    }
}
