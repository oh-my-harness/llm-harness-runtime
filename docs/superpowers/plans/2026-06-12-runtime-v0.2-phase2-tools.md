use llm_harness_runtime::mcp::{McpClientHandle, McpContent, McpError, McpResult, McpToolDef};
use llm_harness_runtime::prompt::{ArgumentDef, PromptSource, PromptSourceTemplate};
use llm_harness_runtime::tool_registry::{ToolDefinition, ToolExecutionMode, ToolSource};
use llm_harness_runtime::error::Result;
use llm_harness_types::{ContentBlock, Tool, ToolContext, ToolError, ToolExecutionMode as CoreExecMode, ToolResult};
use futures::future::BoxFuture;
use serde_json::Value;

/// Adapts an McpClientHandle as a ToolSource.
///
/// Discovers tools from an MCP server via `tools/list` and produces
/// `ToolDefinition` values that the registry can register.
pub struct McpToolSource {
    client: Arc<dyn McpClientHandle>,
}

impl McpToolSource {
    pub fn new(client: Arc<dyn McpClientHandle>) -> Self {
        Self { client }
    }

    /// Discover tools and convert them directly into McpToolAdapter instances.
    ///
    /// Use this instead of manually calling discover() + ToolSourceAdapter,
    /// because McpToolAdapter holds the client ref and can actually call_tool().
    pub async fn into_adapters(&self) -> Result<Vec<McpToolAdapter>> {
        let tools = self.client.list_tools().await?;
        let adapters = tools
            .into_iter()
            .map(|t| McpToolAdapter {
                name: t.name.clone(),
                description: t.description.clone(),
                parameters_schema: t.input_schema.clone(),
                client: self.client.clone(),
            })
            .collect();
        Ok(adapters)
    }
}

impl ToolSource for McpToolSource {
    async fn discover(&self) -> Result<Vec<ToolDefinition>> {
        let tools = self.client.list_tools().await?;
        let definitions = tools
            .into_iter()
            .map(mcp_tool_to_definition)
            .collect();
        Ok(definitions)
    }
}

/// Adapts a single MCP tool + client into a core Tool trait object.
///
/// Unlike ToolSourceAdapter (which only stores the definition),
/// McpToolAdapter holds the McpClientHandle and forwards execute() to call_tool().
pub struct McpToolAdapter {
    name: String,
    description: String,
    parameters_schema: Value,
    client: Arc<dyn McpClientHandle>,
}

impl Tool for McpToolAdapter {
    fn name(&self) -> &str { &self.name }
    fn label(&self) -> &str { &self.name }
    fn description(&self) -> &str { &self.description }
    fn parameters_schema(&self) -> &Value { &self.parameters_schema }
    fn execution_mode(&self) -> CoreExecMode { CoreExecMode::Parallel }

    fn execute<'a>(
        &'a self,
        args: Value,
        _ctx: &'a ToolContext,
    ) -> BoxFuture<'a, std::result::Result<ToolResult, ToolError>> {
        Box::pin(async move {
            match self.client.call_tool(&self.name, args).await {
                Ok(mcp_result) => {
                    let content = mcp_result
                        .content
                        .into_iter()
                        .filter_map(|c| c.text.map(|t| ContentBlock::Text { text: t }))
                        .collect();
                    Ok(ToolResult {
                        content,
                        details: Value::Null,
                        terminate: false,
                    })
                }
                Err(McpError::Protocol { source }) => {
                    Err(ToolError::ExecutionFailed(source.to_string()))
                }
                Err(McpError::ToolExecution { content }) => {
                    // Tool execution error — return as error result so agent can retry.
                    let text = content
                        .into_iter()
                        .filter_map(|c| c.text)
                        .collect::<Vec<_>>()
                        .join("\n");
                    Ok(ToolResult {
                        content: vec![ContentBlock::Text { text }],
                        details: Value::Null,
                        terminate: false,
                    })
                }
            }
        })
    }
}

/// Convert an MCP tool definition to a runtime ToolDefinition (for registry metadata).
fn mcp_tool_to_definition(tool: McpToolDef) -> ToolDefinition {
    ToolDefinition {
        name: tool.name,
        description: tool.description,
        parameters_schema: tool.input_schema,
        execution_mode: ToolExecutionMode::Parallel,
        custom_converter: None,
    }
}

use std::collections::HashMap;
use std::sync::RwLock;

/// In-memory tool registry — the default ToolRegistry implementation.
///
/// Thread-safe via RwLock. Suitable for single-process use.
pub struct InMemoryToolRegistry {
    tools: RwLock<HashMap<String, Arc<dyn Tool>>>,
}

impl InMemoryToolRegistry {
    pub fn new() -> Self {
        Self {
            tools: RwLock::new(HashMap::new()),
        }
    }
}

impl Default for InMemoryToolRegistry {
    fn default() -> Self {
        Self::new()
    }
}

impl ToolRegistry for InMemoryToolRegistry {
    fn register(&self, tool: Arc<dyn Tool>) {
        let mut tools = self.tools.write().unwrap();
        tools.insert(tool.name().to_string(), tool);
    }

    fn unregister(&self, name: &str) {
        let mut tools = self.tools.write().unwrap();
        tools.remove(name);
    }

    fn get(&self, name: &str) -> Option<Arc<dyn Tool>> {
        let tools = self.tools.read().unwrap();
        tools.get(name).cloned()
    }

    fn list(&self) -> Vec<ToolMetadata> {
        let tools = self.tools.read().unwrap();
        tools
            .values()
            .map(|t| ToolMetadata {
                name: t.name().to_string(),
                label: t.label().to_string(),
                description: t.description().to_string(),
            })
            .collect()
    }

    fn subset(&self, names: &[&str]) -> Vec<Arc<dyn Tool>> {
        let tools = self.tools.read().unwrap();
        names
            .iter()
            .filter_map(|name| tools.get(*name).cloned())
            .collect()
    }
}

use std::sync::Arc;

use futures::future::BoxFuture;
use llm_harness_types::{ContentBlock, Tool, ToolContext, ToolError, ToolExecutionMode, ToolResult};
use serde_json::Value;

use crate::tool_registry::ToolDefinition;

/// Adapts a ToolDefinition (from a ToolSource) into a core Tool trait object.
///
/// For MCP-originated tools, prefer `McpToolAdapter` from the `mcp` crate,
/// which holds the McpClientHandle and can dispatch call_tool() on execute().
///
/// Use ToolSourceAdapter only for non-MCP tools where a custom executor
/// is provided via `ToolDefinition::custom_converter`.
pub struct ToolSourceAdapter {
    definition: ToolDefinition,
}

impl ToolSourceAdapter {
    /// Create a new adapter from a ToolDefinition.
    pub fn from_definition(definition: ToolDefinition) -> Self {
        Self { definition }
    }
}

impl Tool for ToolSourceAdapter {
    fn name(&self) -> &str {
        &self.definition.name
    }

    fn label(&self) -> &str {
        &self.definition.name // Use name as label for externally-sourced tools
    }

    fn description(&self) -> &str {
        &self.definition.description
    }

    fn parameters_schema(&self) -> &Value {
        &self.definition.parameters_schema
    }

    fn execution_mode(&self) -> ToolExecutionMode {
        match self.definition.execution_mode {
            crate::tool_registry::ToolExecutionMode::Parallel => ToolExecutionMode::Parallel,
            crate::tool_registry::ToolExecutionMode::Sequential => ToolExecutionMode::Sequential,
        }
    }

    fn execute<'a>(
        &'a self,
        args: Value,
        _ctx: &'a ToolContext,
    ) -> BoxFuture<'a, std::result::Result<ToolResult, ToolError>> {
        Box::pin(async move {
            // Apply custom converter if present (e.g., for legacy parameter formats).
            let converted_args = if let Some(ref converter) = self.definition.custom_converter {
                converter(args).map_err(|msg| ToolError::InvalidArguments(msg))?
            } else {
                args
            };

            // ToolSourceAdapter is for non-MCP tools with a custom_converter.
            // For MCP tools, use McpToolAdapter (in the mcp crate) instead —
            // it holds McpClientHandle and dispatches call_tool() on execute().
            Ok(ToolResult {
                content: vec![ContentBlock::Text {
                    text: format!(
                        "Tool '{}' called with args: {}",
                        self.definition.name,
                        serde_json::to_string(&converted_args).unwrap_or_default()
                    ),
                }],
                details: converted_args,
                terminate: false,
            })
        })
    }
}

use std::collections::HashMap;
use std::sync::RwLock;

use llm_harness::PromptTemplate;

use crate::error::Result;

/// Registy for compiled core PromptTemplates.
///
/// PromptSource templates are discovered, compiled via PromptTemplateCompiler,
/// and stored here. The registry maps template name → compiled PromptTemplate.
pub struct InMemoryPromptRegistry {
    templates: RwLock<HashMap<String, PromptTemplate>>,
}

impl InMemoryPromptRegistry {
    pub fn new() -> Self {
        Self {
            templates: RwLock::new(HashMap::new()),
        }
    }

    /// Register a compiled template.
    pub fn register(&self, name: String, template: PromptTemplate) {
        let mut templates = self.templates.write().unwrap();
        templates.insert(name, template);
    }

    /// Look up a template by name.
    pub fn get(&self, name: &str) -> Option<PromptTemplate> {
        let templates = self.templates.read().unwrap();
        templates.get(name).cloned()
    }

    /// List all registered template names.
    pub fn list_names(&self) -> Vec<String> {
        let templates = self.templates.read().unwrap();
        templates.keys().cloned().collect()
    }

    /// Discover templates from a PromptSource, compile them, and register.
    pub async fn ingest(&self, source: &dyn PromptSource) -> Result<Vec<String>> {
        let raw_templates = source.discover().await?;
        let mut names = Vec::new();
        for raw in &raw_templates {
            let compiled = PromptTemplateCompiler::compile(raw)
                .map_err(|e| anyhow::anyhow!("compile failed for '{}': {e}", raw.name))?;
            self.register(raw.name.clone(), compiled);
            names.push(raw.name.clone());
        }
        Ok(names)
    }

    /// Invoke a registered template with positional arguments.
    pub fn invoke(&self, name: &str, args: &[&str]) -> std::result::Result<String, InvokeError> {
        let template = self
            .get(name)
            .ok_or_else(|| InvokeError::NotFound(name.into()))?;
        template
            .invoke(args)
            .map_err(|e| InvokeError::InvokeFailed(e.to_string()))
    }
}

#[derive(Debug, thiserror::Error)]
pub enum InvokeError {
    #[error("template not found: {0}")]
    NotFound(String),
    #[error("template invocation failed: {0}")]
    InvokeFailed(String),
}

impl Default for InMemoryPromptRegistry {
    fn default() -> Self {
        Self::new()
    }
}
