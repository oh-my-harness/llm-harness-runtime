```rust
/// In-memory TraceExporter for unit/integration testing.
///
/// Accumulates SpanEvent values for assertions. Does NOT connect to any
/// external OTel collector. For production use, provide an OtlpExporter
/// (to be implemented by the business layer via opentelemetry-otlp crate).
pub struct InMemoryTraceExporter {
    /// Accumulated spans for testing/verification.
    exported_spans: Mutex<Vec<SpanEvent>>,

impl InMemoryTraceExporter {
    pub fn new() -> Self {
        Self {
            exported_spans: Mutex::new(Vec::new()),
        }
    }

    /// Return all exported spans (for testing/verification).
    pub fn exported_spans(&self) -> Vec<SpanEvent> {
        self.exported_spans.lock().unwrap().clone()
    }
}

impl Default for InMemoryTraceExporter {
    fn default() -> Self {
        Self::new()
    }
}

impl TraceExporter for InMemoryTraceExporter {
    async fn export(&self, events: &[SpanEvent]) -> Result<()> {
        let mut all = self.exported_spans.lock().unwrap();
        all.extend(events.iter().cloned());
        Ok(())
    }

    async fn flush(&self) -> Result<()> {
        Ok(())
    }
}
```

```rust
use std::collections::HashMap;
use std::sync::{Arc, Mutex};

use chrono::Utc;
use llm_harness::{AfterProviderResponseHook, AfterToolCallHook, AfterTurnHook, BeforeProviderRequestHook};
use llm_harness_types::{
    AfterToolCallCtx, AfterToolCallDecision, AfterTurnCtx,
    BeforeProviderRequestInfo, ProviderResponseInfo,
};

use crate::tracer::{
    AttributeValue, CaptureLevel, SpanCaptureConfig, SpanEvent, SpanId, SpanKind, SpanStatus,
    TraceExporter, TraceId,
};

/// Internal span context tracking — maps turn_index and tool_use_id to active spans.
struct SpanContextMap {
    /// Active task span (one per task).
    task_span: Option<SpanState>,
    task_id: Option<String>,
    /// Active session span (one per harness session).
    session_span: Option<SpanState>,
    session_id: Option<String>,
    model: Option<String>,
    /// Active turn spans, keyed by turn_index.
    turn_spans: HashMap<u32, SpanState>,
    /// LLM call spans started by BeforeProviderRequestHook, keyed by turn_index.
    /// AfterProviderResponseHook removes the entry and uses its start_time.
    llm_spans: HashMap<u32, SpanState>,
    /// Active tool spans, keyed by tool_use_id.
    tool_spans: HashMap<String, SpanState>,
    /// All completed spans (awaiting export).
    completed: Vec<SpanEvent>,
    /// Trace ID shared across all spans in this task.
    trace_id: TraceId,
    /// Current turn index — set by begin_turn(), used by AfterProviderResponseHook.
    current_turn_index: Option<u32>,
}

struct SpanState {
    span_id: SpanId,
    start_time: chrono::DateTime<Utc>,
    parent_span_id: Option<SpanId>,
}

impl SpanContextMap {
    fn new() -> Self {
        Self {
            task_span: None,
            task_id: None,
            session_span: None,
            session_id: None,
            model: None,
            turn_spans: HashMap::new(),
            llm_spans: HashMap::new(),
            tool_spans: HashMap::new(),
            completed: Vec::new(),
            trace_id: TraceId(uuid::Uuid::new_v4()),
            current_turn_index: None,
        }
    }

    fn complete_span(
        &mut self,
        state: SpanState,
        name: String,
        kind: SpanKind,
        attributes: HashMap<String, AttributeValue>,
        status: SpanStatus,
    ) {
        self.completed.push(SpanEvent {
            span_id: state.span_id,
            trace_id: self.trace_id,
            parent_span_id: state.parent_span_id,
            name,
            kind,
            start_time: state.start_time,
            end_time: Utc::now(),
            attributes,
            status,
        });
    }

    fn drain_completed(&mut self) -> Vec<SpanEvent> {
        std::mem::take(&mut self.completed)
    }
}

/// Multi-hook adapter that converts core hook callbacks into span events.
///
/// Implements:
/// - `BeforeProviderRequestHook` → records LLM span start_time in active_spans
/// - `AfterProviderResponseHook` → completes LLM span using stored start_time
/// - `AfterToolCallHook`         → Tool span
/// - `AfterTurnHook`             → completes Turn span using begin_turn() start_time
///
/// Session and Task spans are driven by explicit begin_*/end_* methods called
/// from TaskRunner — NOT from core hooks (they have no paired begin/end events).
pub struct TracingHookAdapter {
    exporter: Arc<dyn TraceExporter>,
    capture_config: SpanCaptureConfig,
    active_spans: Mutex<SpanContextMap>,
}

impl TracingHookAdapter {
    /// Create a new adapter.
    pub fn new(exporter: Arc<dyn TraceExporter>, capture_config: SpanCaptureConfig) -> Self {
        Self {
            exporter,
            capture_config,
            active_spans: Mutex::new(SpanContextMap::new()),
        }
    }

    /// Start a task span (called from TaskRunner).
    /// Stores span state in active_spans — does NOT push to completed.
    /// end_task() must be called to finalize and export the span.
    pub fn begin_task(&self, task_id: &str) {
        let mut spans = self.active_spans.lock().unwrap();
        spans.task_span = Some(SpanState {
            span_id: SpanId(uuid::Uuid::new_v4()),
            start_time: Utc::now(),
            parent_span_id: None,
        });
        // Store task_id for end_task() — here we stash it in the trace context.
        // (Implementation: add `task_id: Option<String>` to SpanContextMap)
        spans.task_id = Some(task_id.to_owned());
    }

    /// End the task span (called from TaskRunner when task finishes/fails).
    /// Sets end_time = now and exports the span.
    pub fn end_task(&self, task_id: &str, status: SpanStatus) {
        let mut spans = self.active_spans.lock().unwrap();
        if let Some(state) = spans.task_span.take() {
            let mut attrs = std::collections::HashMap::new();
            attrs.insert("agent.task_id".into(), AttributeValue::String(task_id.into()));
            spans.complete_span(state, "task".into(), SpanKind::Internal, attrs, status);
        }
        // Span sits in completed until the next exporter drain (async hook or flush()).
    }

    /// Start a session span (called from TaskRunner when AgentHarness starts).
    /// Does NOT push to completed — end_session() finalizes it.
    pub fn begin_session(&self, session_id: &str, model: &str) {
        let mut spans = self.active_spans.lock().unwrap();
        let task_span_id = spans.task_span.as_ref().map(|s| s.span_id);
        spans.session_span = Some(SpanState {
            span_id: SpanId(uuid::Uuid::new_v4()),
            start_time: Utc::now(),
            parent_span_id: task_span_id,
        });
        spans.session_id = Some(session_id.to_owned());
        spans.model = Some(model.to_owned());
    }

    /// End the session span (called from TaskRunner when AgentHarness finishes).
    pub fn end_session(&self, session_id: &str, status: SpanStatus) {
        let mut spans = self.active_spans.lock().unwrap();
        if let Some(state) = spans.session_span.take() {
            let mut attrs = std::collections::HashMap::new();
            attrs.insert("agent.session_id".into(), AttributeValue::String(session_id.into()));
            if let Some(ref m) = spans.model.clone() {
                attrs.insert("gen_ai.model".into(), AttributeValue::String(m.clone()));
            }
            spans.complete_span(state, "session".into(), SpanKind::Internal, attrs, status);
        }
    }

    /// Start a turn span (called before each turn, from BeforeTurnHook or TaskRunner).
    /// Stores start_time in active_spans.turn_spans; AfterTurnHook finalizes it.
    pub fn begin_turn(&self, turn_index: u32) {
        let mut spans = self.active_spans.lock().unwrap();
        spans.current_turn_index = Some(turn_index);
        let session_span_id = spans.session_span.as_ref().map(|s| s.span_id);
        let span_state = SpanState {
            span_id: SpanId(uuid::Uuid::new_v4()),
            start_time: Utc::now(),
            parent_span_id: session_span_id,
        };
        spans.turn_spans.insert(turn_index, span_state);
    }
}

impl BeforeProviderRequestHook for TracingHookAdapter {
    fn before_request<'a>(
        &'a self,
        _info: &'a BeforeProviderRequestInfo,
    ) -> std::pin::Pin<Box<dyn std::future::Future<Output = ()> + Send + 'a>> {
        Box::pin(async move {
            let mut spans = self.active_spans.lock().unwrap();
            if let Some(turn_idx) = spans.current_turn_index {
                let turn_parent = spans.turn_spans.get(&turn_idx).map(|s| s.span_id);
                // Record LLM span start_time = request dispatch time.
                // AfterProviderResponseHook removes this entry and uses its start_time.
                spans.llm_spans.insert(turn_idx, SpanState {
                    span_id: SpanId(uuid::Uuid::new_v4()),
                    start_time: Utc::now(),
                    parent_span_id: turn_parent,
                });
            }
        })
    }
}

impl AfterProviderResponseHook for TracingHookAdapter {
    fn after_response<'a>(
        &'a self,
        info: &'a ProviderResponseInfo,
    ) -> std::pin::Pin<Box<dyn std::future::Future<Output = ()> + Send + 'a>> {
        Box::pin(async move {
            let mut spans = self.active_spans.lock().unwrap();

            // Build attributes from provider response info.
            let mut attrs = HashMap::new();
            attrs.insert(
                "gen_ai.usage.input_tokens".into(),
                AttributeValue::Int(info.usage.as_ref().map(|u| u.input_tokens as i64).unwrap_or(0)),
            );
            attrs.insert(
                "gen_ai.usage.output_tokens".into(),
                AttributeValue::Int(info.usage.as_ref().map(|u| u.output_tokens as i64).unwrap_or(0)),
            );
            attrs.insert(
                "agent.latency_ms".into(),
                AttributeValue::Int(info.latency_ms as i64),
            );

            if let Some(ref usage) = info.usage {
                attrs.insert(
                    "gen_ai.usage.cache_read_tokens".into(),
                    AttributeValue::Int(usage.cache_read_tokens as i64),
                );
                attrs.insert(
                    "gen_ai.usage.cache_creation_tokens".into(),
                    AttributeValue::Int(usage.cache_creation_tokens as i64),
                );
            }

            if let Some(ref status_code) = info.status_code {
                attrs.insert(
                    "http.status_code".into(),
                    AttributeValue::Int(*status_code as i64),
                );
            }

            // Full text capture when configured.
            if self.capture_config.level == CaptureLevel::Full {
                for (key, value) in &info.response_headers {
                    attrs.insert(
                        format!("http.response.header.{key}"),
                        AttributeValue::String(value.clone()),
                    );
                }
            }

            // Pop the LLM span started by BeforeProviderRequestHook.
            // Its start_time is request dispatch time — NOT response arrival time.
            let current_turn = spans.current_turn_index;
            let llm_state = current_turn
                .and_then(|idx| spans.llm_spans.remove(&idx))
                .unwrap_or_else(|| {
                    // Fallback: BeforeProviderRequestHook didn't fire (e.g. in unit tests).
                    let turn_parent = current_turn
                        .and_then(|idx| spans.turn_spans.get(&idx))
                        .map(|s| s.span_id);
                    SpanState {
                        span_id: SpanId(uuid::Uuid::new_v4()),
                        start_time: Utc::now(),
                        parent_span_id: turn_parent,
                    }
                });

            spans.complete_span(
                llm_state,
                "llm_call".into(),
                SpanKind::Client,
                attrs,
                SpanStatus::Ok,
            );

            // Export completed spans.
            let completed = spans.drain_completed();
            drop(spans);

            if !completed.is_empty() {
                let _ = self.exporter.export(&completed).await;
            }
        })
    }
}

impl AfterToolCallHook for TracingHookAdapter {
    fn on_complete<'a>(
        &'a self,
        ctx: AfterToolCallCtx<'a>,
    ) -> std::pin::Pin<Box<dyn std::future::Future<Output = AfterToolCallDecision> + Send + 'a>> {
        Box::pin(async move {
            let mut spans = self.active_spans.lock().unwrap();

            let is_error = ctx.result.is_err();
            let mut attrs = HashMap::from([
                (
                    "agent.tool_name".into(),
                    AttributeValue::String(ctx.tool_name.to_string()),
                ),
                (
                    "agent.tool_use_id".into(),
                    AttributeValue::String(ctx.tool_use_id.to_string()),
                ),
                (
                    "agent.tool_is_error".into(),
                    AttributeValue::Bool(is_error),
                ),
                (
                    "agent.turn_index".into(),
                    AttributeValue::Int(ctx.turn_index as i64),
                ),
            ]);

            if is_error {
                if let Err(ref e) = ctx.result {
                    attrs.insert(
                        "error.message".into(),
                        AttributeValue::String(e.to_string()),
                    );
                }
            }

            // Parent under the current turn span using ctx.turn_index.
            let turn_parent = spans
                .turn_spans
                .get(&ctx.turn_index)
                .map(|s| s.span_id);

            let tool_state = SpanState {
                span_id: SpanId(uuid::Uuid::new_v4()),
                start_time: Utc::now(),
                parent_span_id: turn_parent,
            };

            spans.complete_span(
                tool_state,
                format!("tool.{}", ctx.tool_name),
                SpanKind::Internal,
                attrs,
                if is_error {
                    SpanStatus::Error(ctx.result.as_ref().err().unwrap().to_string())
                } else {
                    SpanStatus::Ok
                },
            );

            let completed = spans.drain_completed();
            drop(spans);

            if !completed.is_empty() {
                let _ = self.exporter.export(&completed).await;
            }

            AfterToolCallDecision::Passthrough
        })
    }
}

impl AfterTurnHook for TracingHookAdapter {
    fn after_turn<'a>(
        &'a self,
        ctx: AfterTurnCtx<'a>,
    ) -> std::pin::Pin<Box<dyn std::future::Future<Output = ()> + Send + 'a>> {
        Box::pin(async move {
            let mut spans = self.active_spans.lock().unwrap();

            let attrs = HashMap::from([
                (
                    "agent.turn_index".into(),
                    AttributeValue::Int(ctx.turn_index as i64),
                ),
                (
                    "agent.new_message_count".into(),
                    AttributeValue::Int(ctx.new_messages.len() as i64),
                ),
            ]);

            // Pop the turn span created by begin_turn() — its start_time is the
            // actual start of this turn, NOT the time AfterTurnHook fires.
            let turn_state = spans.turn_spans.remove(&ctx.turn_index).unwrap_or_else(|| {
                // Fallback: begin_turn() was not called (e.g. single-turn tests).
                SpanState {
                    span_id: SpanId(uuid::Uuid::new_v4()),
                    start_time: Utc::now(),
                    parent_span_id: spans.session_span.as_ref().map(|s| s.span_id),
                }
            });

            spans.complete_span(
                turn_state,
                format!("turn.{}", ctx.turn_index),
                SpanKind::Internal,
                attrs,
                SpanStatus::Ok,
            );

            let completed = spans.drain_completed();
            drop(spans);

            if !completed.is_empty() {
                let _ = self.exporter.export(&completed).await;
            }
        })
    }
}
```
