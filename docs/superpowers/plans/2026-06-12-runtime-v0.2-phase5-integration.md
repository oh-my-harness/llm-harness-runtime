//! Mock LlmClient for integration testing.
//!
//! The mock simulates LLM responses without real API calls.
//! Responses are pre-programmed per test case.

use std::collections::VecDeque;
use std::sync::Mutex;

use futures::future::BoxFuture;
use futures::FutureExt;

/// A controllable mock LLM client for integration testing.
///
/// Responses are enqueued before the test and returned in order.
/// If the queue is empty, the last enqueued "default" response is reused.
pub struct MockLlmClient {
    responses: Mutex<VecDeque<MockResponse>>,
    default_response: MockResponse,
}

/// Pre-programmed response for a single LLM call.
#[derive(Clone)]
pub struct MockResponse {
    /// The text content returned by the mock.
    pub text: String,
    /// Optional tool calls to include in the response.
    pub tool_calls: Vec<MockToolCall>,
    /// Stop reason.
    pub stop_reason: String,
    /// Simulated token usage.
    pub input_tokens: u32,
    pub output_tokens: u32,
}

/// A tool call requested by the mock (simulates LLM tool_use).
#[derive(Clone)]
pub struct MockToolCall {
    pub id: String,
    pub name: String,
    pub arguments: serde_json::Value,
}

impl MockLlmClient {
    pub fn new(default_response: MockResponse) -> Self {
        Self {
            responses: Mutex::new(VecDeque::new()),
            default_response,
        }
    }

    /// Enqueue a response to be returned on the next call.
    pub fn enqueue(&self, response: MockResponse) {
        self.responses.lock().unwrap().push_back(response);
    }

    /// Pop the next response from the queue, or return the default.
    fn next_response(&self) -> MockResponse {
        self.responses
            .lock()
            .unwrap()
            .pop_front()
            .unwrap_or_else(|| self.default_response.clone())
    }
}

/// Build a simple end-turn response (agent finishes).
pub fn end_turn_response(text: &str) -> MockResponse {
    MockResponse {
        text: text.into(),
        tool_calls: vec![],
        stop_reason: "end_turn".into(),
        input_tokens: 100,
        output_tokens: 50,
    }
}

/// Build a tool-call response (agent requests tool execution).
pub fn tool_call_response(tool_name: &str, args: serde_json::Value) -> MockResponse {
    MockResponse {
        text: String::new(),
        tool_calls: vec![MockToolCall {
            id: format!("tool_{}", uuid::Uuid::new_v4()),
            name: tool_name.into(),
            arguments: args,
        }],
        stop_reason: "tool_use".into(),
        input_tokens: 200,
        output_tokens: 30,
    }
}

use std::sync::{Arc, Mutex};

use llm_harness_runtime::budget::{BudgetControlAdapter, BudgetExceededHook};
use llm_harness_runtime::cost::{CostAggregate, PricingProvider, TokenPrice};

    price: TokenPrice,
}

    fn price_for(&self, _model: &str, _provider: &str) -> Option<TokenPrice> {
        Some(self.price.clone())
    }

/// A BudgetExceededHook that records calls.
struct RecordingBudgetHook {
    called: Mutex<bool>,
    response: Mutex<bool>, // true = continue, false = stop
}

impl RecordingBudgetHook {
    fn new(response: bool) -> Self {
        Self {
            called: Mutex::new(false),
            response: Mutex::new(response),
        }
    }

    fn was_called(&self) -> bool {
        *self.called.lock().unwrap()
    }
}

impl BudgetExceededHook for RecordingBudgetHook {
    async fn on_budget_exceeded(
        &self,
        _session_cost: &CostAggregate,
        _task_cost: &CostAggregate,
        _budget_limit: f64,
    ) -> bool {
        *self.called.lock().unwrap() = true;
        *self.response.lock().unwrap()
    }
}

//! Audit log integration tests — full chain from event to file.

use std::sync::Arc;

use chrono::Utc;
use llm_harness_runtime::audit::{
    AuditDecision, AuditEntry, AuditEventType, AuditSink,
};
use llm_harness_types::EntryId;
use llm_harness_runtime_audit_jsonl::JsonlAuditSink;

fn make_entry(trace_id: &str, event_type: AuditEventType, decision: AuditDecision) -> AuditEntry {
    AuditEntry {
        timestamp: Utc::now(),
        trace_id: trace_id.into(),
        session_id: EntryId(uuid::Uuid::new_v4()),
        task_id: Some("task-1".into()),
        principal: "agent".into(),
        event_type,
        payload: serde_json::json!({"note": "test"}),
        decision: Some(decision),
        cost_delta: None,
        prev_hash: None,
        hash: String::new(),
    }
}

/// Test the complete audit chain: write events → flush → validate.
#[tokio::test]
async fn full_audit_chain_write_and_validate() {
    let dir = tempfile::tempdir().unwrap();
    let path = dir.path().join("full_chain.jsonl");
    let sink = JsonlAuditSink::new(&path);

    // Write a realistic sequence of audit events.
    sink.record(make_entry("trace-1", AuditEventType::Auth, AuditDecision::Allow))
        .await
        .unwrap();
    sink.record(make_entry(
        "trace-1",
        AuditEventType::StateTransition,
        AuditDecision::Allow,
    ))
    .await
    .unwrap();
    sink.record(make_entry(
        "trace-1",
        AuditEventType::ToolCall,
        AuditDecision::Allow,
    ))
    .await
    .unwrap();
    sink.record(make_entry(
        "trace-2",
        AuditEventType::HumanApproval,
        AuditDecision::Deny {
            reason: "user rejected".into(),
        },
    ))
    .await
    .unwrap();
    sink.record(make_entry(
        "trace-2",
        AuditEventType::BudgetExceeded,
        AuditDecision::Deny {
            reason: "limit reached".into(),
        },
    ))
    .await
    .unwrap();
    sink.flush().await.unwrap();

    // Validate hash chain.
    let count = JsonlAuditSink::validate(&path).await.unwrap();
    assert_eq!(count, 5);

    // Verify file contents are valid JSONL.
    let content = tokio::fs::read_to_string(&path).await.unwrap();
    let lines: Vec<&str> = content.lines().filter(|l| !l.is_empty()).collect();
    assert_eq!(lines.len(), 5);

    for (i, line) in lines.iter().enumerate() {
        let parsed: serde_json::Value =
            serde_json::from_str(line).unwrap_or_else(|e| panic!("line {i} not valid JSON: {e}"));
        assert!(parsed.get("hash").is_some(), "line {i} missing hash");
        assert!(parsed.get("timestamp").is_some(), "line {i} missing timestamp");
        assert!(parsed.get("event_type").is_some(), "line {i} missing event_type");
    }
}

/// Each entry should chain to the previous via prev_hash.
#[tokio::test]
async fn hash_chain_is_sequential() {
    let dir = tempfile::tempdir().unwrap();
    let path = dir.path().join("chain.jsonl");
    let sink = JsonlAuditSink::new(&path);

    sink.record(make_entry("trace-1", AuditEventType::Auth, AuditDecision::Allow))
        .await
        .unwrap();
    sink.record(make_entry("trace-1", AuditEventType::ToolCall, AuditDecision::Allow))
        .await
        .unwrap();
    sink.flush().await.unwrap();

    let content = tokio::fs::read_to_string(&path).await.unwrap();
    let lines: Vec<serde_json::Value> = content
        .lines()
        .filter(|l| !l.is_empty())
        .map(|l| serde_json::from_str(l).unwrap())
        .collect();

    assert_eq!(lines.len(), 2);

    // First entry: prev_hash should be null/empty.
    let prev0 = lines[0].get("prev_hash").and_then(|v| v.as_str());
    assert!(prev0.is_none() || prev0 == Some(""));

    // Second entry: prev_hash should equal first entry's hash.
    let hash0 = lines[0]["hash"].as_str().unwrap();
    let prev1 = lines[1]["prev_hash"].as_str().unwrap();
    assert_eq!(hash0, prev1, "prev_hash should chain to previous entry's hash");
}

/// Corrupted hash (content modified but hash not recomputed) is detected.
#[tokio::test]
async fn corrupted_content_breaks_hash_chain() {
    let dir = tempfile::tempdir().unwrap();
    let path = dir.path().join("corrupt.jsonl");
    let sink = JsonlAuditSink::new(&path);

    sink.record(make_entry("trace-1", AuditEventType::Auth, AuditDecision::Allow))
        .await
        .unwrap();
    sink.record(make_entry("trace-1", AuditEventType::ToolCall, AuditDecision::Allow))
        .await
        .unwrap();
    sink.flush().await.unwrap();

    // Corrupt the second line: change Allow to Deny without recomputing hash.
    let mut content = tokio::fs::read_to_string(&path).await.unwrap();
    content = content.replace("\"decision\":\"Allow\"", "\"decision\":\"Deny\"");
    tokio::fs::write(&path, content).await.unwrap();

    let result = JsonlAuditSink::validate(&path).await;
    assert!(result.is_err(), "Tampered log should fail validation");
}

    sandbox.shutdown().await.unwrap();
